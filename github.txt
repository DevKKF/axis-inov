# Assure-toi d'être sur la branche develop
git checkout develop

# Récupère les dernières infos du dépôt distant
git fetch origin

# Remplace TOUT le contenu local par celui de la branche distante
git reset --hard origin/develop

# Récupérer les changements effectués
git pull origin develop

#Liste des branches distantes :
git branch -r

#Ouverture du dossier d'accès pour exécution de commande :
cd src

#Commande de récupération du css et js :
python manage.py collectstatic

#Commande migration :
python manage.py migrate

#Commande seeders :
python manage.py seed_all
